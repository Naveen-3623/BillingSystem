Bill Table:

CREATE TABLE BILL_TB (
    BILLID VARCHAR2(12) PRIMARY KEY,
    CUSTOMERNAME VARCHAR2(50) NOT NULL,
    PRODUCTNAME VARCHAR2(50) NOT NULL,
    BILL_DATE DATE NOT NULL,
    QUANTITY NUMBER,
    PRICE NUMBER,
    TOTALAMOUNT NUMBER,
    REMARKS VARCHAR2(100)
);

Sequence:

CREATE SEQUENCE BILL_SEQ
START WITH 10
INCREMENT BY 1
MAXVALUE 99;
